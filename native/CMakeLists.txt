cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS version for the player libraries" FORCE)
endif()
project(blackhole LANGUAGES CXX)
add_library(blackhole SHARED blackhole.cpp)
target_compile_features(blackhole PRIVATE cxx_std_17)
set_target_properties(blackhole PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../UnityProject/Assets/Plugins/macOS")
if(APPLE)
 enable_language(OBJCXX)
 add_library(helmetphoto SHARED helmetphoto.mm)
 target_compile_options(helmetphoto PRIVATE -fobjc-arc)
 target_link_libraries(helmetphoto PRIVATE "-framework Cocoa" "-framework UniformTypeIdentifiers" "-framework ImageIO")
 set_target_properties(helmetphoto PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../UnityProject/Assets/Plugins/macOS")
 add_library(facepilot SHARED facepilot.mm)
 target_compile_features(facepilot PRIVATE cxx_std_17)
 target_compile_options(facepilot PRIVATE -fobjc-arc)
 target_link_libraries(facepilot PRIVATE "-framework Cocoa" "-framework AVFoundation" "-framework Vision" "-framework CoreVideo" "-framework CoreMedia" "-framework ImageIO")
 set_target_properties(facepilot PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../UnityProject/Assets/Plugins/macOS")
endif()
